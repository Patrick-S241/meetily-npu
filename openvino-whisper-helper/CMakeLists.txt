cmake_minimum_required(VERSION 3.26)
project(openvino_whisper_helper VERSION 0.1.0 LANGUAGES CXX)

set(OPENVINO_WHISPER_OPENVINO_VERSION "2026.4.0" CACHE STRING "Required OpenVINO Runtime version")
set(OPENVINO_WHISPER_GENAI_VERSION "2026.4.0.0" CACHE STRING "Required OpenVINO GenAI version")
set(OPENVINO_WHISPER_JSON_INCLUDE "" CACHE PATH "Directory containing nlohmann/json.hpp from the pinned GenAI archive")
option(OPENVINO_WHISPER_BUILD_HELPER "Build the OpenVINO GenAI sidecar" ON)
option(OPENVINO_WHISPER_BUILD_TESTS "Build protocol tests without NPU hardware" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT EXISTS "${OPENVINO_WHISPER_JSON_INCLUDE}/nlohmann/json.hpp")
    message(FATAL_ERROR "Set OPENVINO_WHISPER_JSON_INCLUDE to <OpenVINO GenAI>/samples/cpp/thirdparty/nlohmann_json/single_include")
endif()

add_library(openvino_whisper_protocol STATIC src/protocol.cpp)
target_include_directories(openvino_whisper_protocol PUBLIC src)
target_include_directories(openvino_whisper_protocol SYSTEM PUBLIC "${OPENVINO_WHISPER_JSON_INCLUDE}")
target_compile_definitions(openvino_whisper_protocol PUBLIC OPENVINO_WHISPER_PROTOCOL_VERSION=1)

if(OPENVINO_WHISPER_BUILD_HELPER)
    # The archive uses the nonstandard filename OpenVINOConfig-version.cmake;
    # CMake does not load it as a package-version file. Validate it directly.
    find_package(OpenVINO REQUIRED)
    set(_openvino_runtime_version_file "${OpenVINO_DIR}/OpenVINOConfig-version.cmake")
    if(NOT EXISTS "${_openvino_runtime_version_file}")
        message(FATAL_ERROR "Pinned OpenVINO archive is missing ${_openvino_runtime_version_file}")
    endif()
    set(PACKAGE_FIND_VERSION "${OPENVINO_WHISPER_OPENVINO_VERSION}")
    include("${_openvino_runtime_version_file}")
    if(NOT PACKAGE_VERSION STREQUAL OPENVINO_WHISPER_OPENVINO_VERSION)
        message(FATAL_ERROR "OpenVINO Runtime ${OPENVINO_WHISPER_OPENVINO_VERSION} is required; archive reports ${PACKAGE_VERSION}")
    endif()
    find_package(OpenVINOGenAI ${OPENVINO_WHISPER_GENAI_VERSION} EXACT REQUIRED)
    add_executable(openvino-whisper-helper src/main.cpp src/whisper_engine.cpp src/device_probe.cpp)
    target_link_libraries(openvino-whisper-helper PRIVATE openvino_whisper_protocol openvino::runtime openvino::genai)
    target_compile_definitions(openvino-whisper-helper PRIVATE OPENVINO_WHISPER_OPENVINO_VERSION="${OPENVINO_WHISPER_OPENVINO_VERSION}")
endif()

if(OPENVINO_WHISPER_BUILD_TESTS)
    enable_testing()
    add_executable(protocol_tests tests/protocol_tests.cpp)
    target_link_libraries(protocol_tests PRIVATE openvino_whisper_protocol)
    add_test(NAME protocol_tests COMMAND protocol_tests)
endif()
